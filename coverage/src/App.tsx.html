import React, { useCallback } from 'react';
import { useAppDispatch, useAppSelector } from './shared/store/hooks';
import { uiActions } from './shared/store/uiSlice';
import { ApplicantsDashboard } from './features/applicantsDashboard';
import { ApplicationForm } from './features/applicationForm';
import { GraduationCap, LayoutDashboard, FileText, Bell, Menu, X, Plus, Sun, Moon } from 'lucide-react';
import { isEqualNormalized } from './shared/utils/normalizeKey';
 
export const App: React.FC = () => {
  const dispatch = useAppDispatch();
  const currentTab = useAppSelector((s) => s.ui.currentTab);
  const mobileMenuOpen = useAppSelector((s) => s.ui.mobileMenuOpen);
  const theme = useAppSelector((s) => s.ui.theme);
 
  const isDark = isEqualNormalized(theme, 'dark');
 
  const handleNavChange = useCallback(
    (tab: 'dashboard' | 'application') => {
      dispatch(uiActions.setCurrentTab(tab));
      dispatch(uiActions.setMobileMenuOpen(false));
    },
    [dispatch]
  );
 
  const toggleMobileMenu = useCallback(() => {
    dispatch(uiActions.toggleMobileMenu());
  }, [dispatch]);
 
  const toggleTheme = useCallback(() => {
    dispatch(uiActions.toggleTheme());
  }, [dispatch]);
 
  return (
    <div className={`min-h-screen flex flex-col font-sans transition-colors duration-200 ${isDark ? 'dark bg-slate-950 text-slate-100' : 'bg-[#f8f9ff] text-slate-900'}`}>
      {/* Mobile Header */}
      <header className="md:hidden bg-white/80 dark:bg-slate-900/80 backdrop-blur-md border-b border-slate-200/80 dark:border-slate-800 flex justify-between items-center px-4 py-3 z-30 sticky top-0 shadow-xs">
        <div className="flex items-center gap-3">
          <button
            onClick={toggleMobileMenu}
            className="text-slate-600 dark:text-slate-300 p-2 rounded-xl hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
          >
            {mobileMenuOpen ? <X className="w-5 h-5" /> : <Menu className="w-5 h-5" />}
          </button>
          <div className="flex items-center gap-2">
            <div className="w-8 h-8 rounded-lg bg-[#002045] dark:bg-blue-600 text-white flex items-center justify-center font-bold">
              <GraduationCap className="w-5 h-5" />
            </div>
            <h1 className="font-bold text-lg text-[#002045] dark:text-slate-100 tracking-tight">EduPortal</h1>
          </div>
        </div>
        <div className="flex items-center gap-2">
          <button
            onClick={toggleTheme}
            className="text-slate-600 dark:text-slate-300 p-2 rounded-full hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
          >
            {isDark ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
          </button>
          <button className="text-slate-600 dark:text-slate-300 p-2 rounded-full hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors">
            <Bell className="w-5 h-5" />
          </button>
        </div>
      </header>
 
      <div className="flex-1 flex min-w-0">
        {/* Sidebar Navigation */}
        <aside
          className={`${
            mobileMenuOpen ? 'block fixed inset-y-0 left-0 z-40 shadow-2xl bg-white dark:bg-slate-900 w-72' : 'hidden'
          } md:flex bg-white dark:bg-slate-900 border-r border-slate-200/80 dark:border-slate-800 w-64 flex-col p-5 space-y-6 shrink-0`}
        >
          <div className="flex items-center justify-between px-2">
            <div className="flex items-center gap-3">
              <div className="w-10 h-10 rounded-xl bg-gradient-to-tr from-[#002045] to-[#1a365d] dark:from-blue-600 dark:to-blue-700 text-white flex items-center justify-center font-bold shadow-md shadow-[#002045]/10">
                <GraduationCap className="w-6 h-6" />
              </div>
              <div>
                <h1 className="font-bold text-xl text-[#002045] dark:text-slate-100 tracking-tight">EduPortal</h1>
                <p className="text-xs font-semibold text-slate-400 dark:text-slate-500">Admissions Portal</p>
              </div>
            </div>
            <button
              onClick={toggleTheme}
              className="hidden md:flex text-slate-600 dark:text-slate-300 p-2 rounded-xl hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
            >
              {isDark ? <Sun className="w-4 h-4" /> : <Moon className="w-4 h-4" />}
            </button>
          </div>
 
          <button
            onClick={() => handleNavChange('application')}
            className="w-full bg-[#002045] hover:bg-[#1a365d] dark:bg-blue-600 dark:hover:bg-blue-500 text-white text-sm font-semibold py-3 px-4 rounded-xl transition-all shadow-sm hover:shadow-md flex items-center justify-center gap-2 active:scale-[0.98]"
          >
            <Plus className="w-4 h-4" />
            Start Application
          </button>
 
          <nav className="flex-1 space-y-1.5">
            <button
              onClick={() => handleNavChange('dashboard')}
              className={`w-full flex items-center gap-3 px-4 py-3 text-sm font-semibold rounded-xl transition-all ${
                currentTab === 'dashboard'
                  ? 'bg-slate-100 dark:bg-slate-800 text-[#002045] dark:text-blue-400 shadow-xs'
                  : 'text-slate-600 dark:text-slate-400 hover:bg-slate-50 dark:hover:bg-slate-800/50 hover:text-slate-900 dark:hover:text-slate-200'
              }`}
            >
              <LayoutDashboard className="w-4 h-4" />
              <span>Dashboard</span>
            </button>
 
            <button
              onClick={() => handleNavChange('application')}
              className={`w-full flex items-center gap-3 px-4 py-3 text-sm font-semibold rounded-xl transition-all ${
                currentTab === 'application'
                  ? 'bg-slate-100 dark:bg-slate-800 text-[#002045] dark:text-blue-400 shadow-xs'
                  : 'text-slate-600 dark:text-slate-400 hover:bg-slate-50 dark:hover:bg-slate-800/50 hover:text-slate-900 dark:hover:text-slate-200'
              }`}
            >
              <FileText className="w-4 h-4" />
              <span>Application Form</span>
            </button>
          </nav>
        </aside>
 
        {/* Content Area */}
        <main className="flex-1 overflow-y-auto p-4 sm:p-8">
          <div className="max-w-6xl mx-auto">
            {currentTab === 'dashboard' ? <ApplicantsDashboard /> : <ApplicationForm />}
          </div>
        </main>
      </div>
 
      {/* Footer */}
      <footer className="bg-white dark:bg-slate-900 border-t border-slate-200/80 dark:border-slate-800 py-4 px-6 mt-auto">
        <div className="max-w-6xl mx-auto flex flex-col sm:flex-row justify-between items-center text-xs text-slate-500 dark:text-slate-400 gap-2">
          <span>© 2026 EduPortal University Admissions. All rights reserved.</span>
          <div className="flex gap-4">
            <a href="#" className="hover:text-[#002045] dark:hover:text-slate-200 underline transition-colors">
              Privacy Policy
            </a>
            <a href="#" className="hover:text-[#002045] dark:hover:text-slate-200 underline transition-colors">
              Terms of Service
            </a>
            <a href="#" className="hover:text-[#002045] dark:hover:text-slate-200 underline transition-colors">
              Contact Support
            </a>
          </div>
        </div>
      </footer>
    </div>
  );
};
