import React, { useCallback, useEffect } from 'react';
import { useAppDispatch, useAppSelector } from '../../shared/store/hooks';
import { applicantsActions } from './slice';
import { uiActions } from '../../shared/store/uiSlice';
import { isEqualNormalized } from '../../shared/utils/normalizeKey';
import { Search, RefreshCw, AlertCircle, SearchX, Plus, Eye, MoreVertical, Clock, CheckCircle2, FileText, ArrowUpDown, Globe, Building } from 'lucide-react';
import type { Applicant } from './types';
 
export const ApplicantsDashboard: React.FC = () => {
  const dispatch = useAppDispatch();
  const items = useAppSelector((s) => s.applicants.items as Applicant[]);
  const status = useAppSelector((s) => s.applicants.status);
  const error = useAppSelector((s) => s.applicants.error);
  const searchQuery = useAppSelector((s) => s.applicants.searchQuery);
  const sortBy = useAppSelector((s) => s.applicants.sortBy);
  const sortOrder = useAppSelector((s) => s.applicants.sortOrder);
 
  const fetchItems = useCallback(() => {
    dispatch(applicantsActions.fetchRequest(undefined));
  }, [dispatch]);
 
  useEffect(() => {
    if (isEqualNormalized(status, 'idle')) {
      fetchItems();
    }
  }, [status, fetchItems]);
 
  const handleSearchChange = useCallback(
    (e: React.ChangeEvent<HTMLInputElement>) => {
      dispatch(applicantsActions.setSearchQuery(e.target.value));
    },
    [dispatch]
  );
 
  const handleSort = useCallback(
    (column: string) => {
      dispatch(applicantsActions.setSort({ sortBy: column }));
    },
    [dispatch]
  );
 
  const handleStartNewApp = useCallback(() => {
    dispatch(uiActions.setCurrentTab('application'));
  }, [dispatch]);
 
  const filteredItems = React.useMemo(() => {
    let result = [...items];
 
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      result = result.filter(
        (item) =>
          item.name.toLowerCase().includes(q) ||
          item.username.toLowerCase().includes(q) ||
          item.email.toLowerCase().includes(q) ||
          item.phone.includes(q) ||
          (item.website && item.website.toLowerCase().includes(q)) ||
          (item.company?.name && item.company.name.toLowerCase().includes(q)) ||
          item.status.toLowerCase().includes(q)
      );
    }
 
    if (sortBy) {
      result.sort((a, b) => {
        let valA = '';
        let valB = '';
        if (isEqualNormalized(sortBy, 'name')) {
          valA = a.name.toLowerCase();
          valB = b.name.toLowerCase();
        } else if (isEqualNormalized(sortBy, 'status')) {
          valA = a.status.toLowerCase();
          valB = b.status.toLowerCase();
        } else if (isEqualNormalized(sortBy, 'email')) {
          valA = a.email.toLowerCase();
          valB = b.email.toLowerCase();
        } else if (isEqualNormalized(sortBy, 'company')) {
          valA = (a.company?.name || '').toLowerCase();
          valB = (b.company?.name || '').toLowerCase();
        }
 
        if (valA < valB) return sortOrder === 'asc' ? -1 : 1;
        if (valA > valB) return sortOrder === 'asc' ? 1 : -1;
        return 0;
      });
    }
 
    return result;
  }, [items, searchQuery, sortBy, sortOrder]);
 
  const getStatusBadge = (appStatus: Applicant['status']) => {
    if (isEqualNormalized(appStatus, 'Approved')) {
      return (
        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-semibold bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300 border border-emerald-200/60 dark:border-emerald-800/60 shadow-xs">
          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 animate-pulse"></span>
          Approved
        </span>
      );
    }
    if (isEqualNormalized(appStatus, 'Under Review')) {
      return (
        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-semibold bg-amber-50 text-amber-800 dark:bg-amber-950/40 dark:text-amber-300 border border-amber-200/60 dark:border-amber-800/60 shadow-xs">
          <span className="w-1.5 h-1.5 rounded-full bg-amber-500"></span>
          Under Review
        </span>
      );
    }
    if (isEqualNormalized(appStatus, 'Pending')) {
      return (
        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-semibold bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300 border border-slate-200 dark:border-slate-700 shadow-xs">
          <span className="w-1.5 h-1.5 rounded-full bg-slate-400"></span>
          Pending
        </span>
      );
    }
    if (isEqualNormalized(appStatus, 'Waitlisted')) {
      return (
        <span className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-semibold bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300 border border-rose-200/60 dark:border-rose-800/60 shadow-xs">
          <span className="w-1.5 h-1.5 rounded-full bg-rose-500"></span>
          Waitlisted
        </span>
      );
    }
    return (
      <span className="inline-flex items-center px-3 py-1 rounded-full text-xs font-medium bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400">
        {appStatus}
      </span>
    );
  };
 
  return (
    <div className="space-y-6">
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 pb-2 border-b border-slate-200/80 dark:border-slate-800">
        <div>
          <h2 className="text-2xl md:text-3xl font-bold tracking-tight text-[#002045] dark:text-slate-100">
            Applications Dashboard
          </h2>
          <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
            Real-time applicant management rendering all fields dynamically from the API (`username`, `website`, `company`).
          </p>
        </div>
        <div className="flex flex-col sm:flex-row gap-3 items-center">
          <div className="relative w-full sm:w-80">
            <div className="absolute inset-y-0 left-0 pl-3.5 flex items-center pointer-events-none text-slate-400">
              <Search className="w-4 h-4" />
            </div>
            <input
              type="text"
              value={searchQuery}
              onChange={handleSearchChange}
              placeholder="Search by name, company, website, status..."
              className="block w-full pl-10 pr-4 py-2.5 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-700 rounded-xl text-sm text-slate-900 dark:text-slate-100 placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-[#002045]/20 focus:border-[#002045] dark:focus:border-blue-400 transition-all shadow-xs"
            />
          </div>
          <button
            onClick={handleStartNewApp}
            className="w-full sm:w-auto bg-[#002045] hover:bg-[#1a365d] dark:bg-blue-600 dark:hover:bg-blue-500 text-white text-sm font-semibold px-5 py-2.5 rounded-xl transition-all shadow-sm hover:shadow-md flex items-center justify-center gap-2 shrink-0 active:scale-[0.98]"
          >
            <Plus className="w-4 h-4" />
            New Application
          </button>
        </div>
      </div>
 
      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
        <div className="bg-white dark:bg-slate-900 border border-slate-200/80 dark:border-slate-800 rounded-2xl p-5 flex items-center justify-between shadow-xs hover:shadow-sm transition-all group">
          <div>
            <p className="text-xs font-semibold text-slate-400 dark:text-slate-500 uppercase tracking-wider">
              Total Applicants
            </p>
            <p className="text-3xl font-bold text-slate-900 dark:text-slate-100 mt-1.5 group-hover:text-[#002045] dark:group-hover:text-blue-400 transition-colors">
              {items.length}
            </p>
          </div>
          <div className="w-12 h-12 rounded-xl bg-blue-50 dark:bg-slate-800 text-[#002045] dark:text-blue-400 flex items-center justify-center shrink-0">
            <FileText className="w-6 h-6" />
          </div>
        </div>
 
        <div className="bg-white dark:bg-slate-900 border border-slate-200/80 dark:border-slate-800 rounded-2xl p-5 flex items-center justify-between shadow-xs hover:shadow-sm transition-all group">
          <div>
            <p className="text-xs font-semibold text-slate-400 dark:text-slate-500 uppercase tracking-wider">
              Under Review
            </p>
            <p className="text-3xl font-bold text-slate-900 dark:text-slate-100 mt-1.5 group-hover:text-amber-600 dark:group-hover:text-amber-400 transition-colors">
              {items.filter((i: Applicant) => isEqualNormalized(i.status, 'Under Review')).length}
            </p>
          </div>
          <div className="w-12 h-12 rounded-xl bg-amber-50 dark:bg-slate-800 text-amber-600 dark:text-amber-400 flex items-center justify-center shrink-0">
            <Clock className="w-6 h-6" />
          </div>
        </div>
 
        <div className="bg-white dark:bg-slate-900 border border-slate-200/80 dark:border-slate-800 rounded-2xl p-5 flex items-center justify-between shadow-xs hover:shadow-sm transition-all group">
          <div>
            <p className="text-xs font-semibold text-slate-400 dark:text-slate-500 uppercase tracking-wider">
              Approved
            </p>
            <p className="text-3xl font-bold text-slate-900 dark:text-slate-100 mt-1.5 group-hover:text-emerald-600 dark:group-hover:text-emerald-400 transition-colors">
              {items.filter((i: Applicant) => isEqualNormalized(i.status, 'Approved')).length}
            </p>
          </div>
          <div className="w-12 h-12 rounded-xl bg-emerald-50 dark:bg-slate-800 text-emerald-600 dark:text-emerald-400 flex items-center justify-center shrink-0">
            <CheckCircle2 className="w-6 h-6" />
          </div>
        </div>
      </div>
 
      <div className="bg-white dark:bg-slate-900 border border-slate-200/80 dark:border-slate-800 rounded-2xl shadow-xs overflow-hidden flex flex-col min-h-[420px]">
        {isEqualNormalized(status, 'loading') && (
          <div className="flex-1 flex flex-col items-center justify-center py-16">
            <RefreshCw className="animate-spin text-[#002045] dark:text-blue-400 w-9 h-9 mb-3" />
            <p className="text-sm font-medium text-slate-500 dark:text-slate-400">Loading applicant data...</p>
          </div>
        )}
 
        {isEqualNormalized(status, 'failed') && (
          <div className="flex-1 flex flex-col items-center justify-center py-16 px-4 text-center">
            <div className="w-14 h-14 rounded-full bg-rose-50 dark:bg-rose-950/40 text-rose-600 dark:text-rose-400 flex items-center justify-center mb-3">
              <AlertCircle className="w-7 h-7" />
            </div>
            <h3 className="text-lg font-semibold text-slate-900 dark:text-slate-100 mb-1">Failed to load data</h3>
            <p className="text-sm text-slate-500 dark:text-slate-400 max-w-sm mb-4">{error}</p>
            <button
              onClick={fetchItems}
              className="px-4 py-2 border border-slate-300 dark:border-slate-700 rounded-xl text-slate-700 dark:text-slate-300 font-medium text-sm hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors"
            >
              Try Again
            </button>
          </div>
        )}
 
        {isEqualNormalized(status, 'succeeded') && filteredItems.length === 0 && (
          <div className="flex-1 flex flex-col items-center justify-center py-16 px-4 text-center">
            <div className="w-14 h-14 rounded-full bg-slate-100 dark:bg-slate-800 text-slate-400 flex items-center justify-center mb-3">
              <SearchX className="w-7 h-7" />
            </div>
            <h3 className="text-lg font-semibold text-slate-900 dark:text-slate-100 mb-1">No applicants found</h3>
            <p className="text-sm text-slate-500 dark:text-slate-400">Try searching for a different keyword or status.</p>
          </div>
        )}
 
        {isEqualNormalized(status, 'succeeded') && filteredItems.length > 0 && (
          <>
            <div className="hidden lg:block overflow-x-auto flex-1">
              <table className="w-full text-left border-collapse">
                <thead>
                  <tr className="bg-slate-50/80 dark:bg-slate-800/60 border-b border-slate-200/80 dark:border-slate-800 text-xs font-semibold text-slate-500 dark:text-slate-400 uppercase tracking-wider">
                    <th
                      className="py-3.5 px-5 cursor-pointer hover:text-slate-900 dark:hover:text-slate-200 transition-colors"
                      onClick={() => handleSort('name')}
                    >
                      <div className="flex items-center gap-1.5">
                        Applicant Name & Handle
                        <ArrowUpDown className="w-3.5 h-3.5 text-slate-400" />
                      </div>
                    </th>
                    <th
                      className="py-3.5 px-5 cursor-pointer hover:text-slate-900 dark:hover:text-slate-200 transition-colors"
                      onClick={() => handleSort('email')}
                    >
                      <div className="flex items-center gap-1.5">
                        Contact Info
                        <ArrowUpDown className="w-3.5 h-3.5 text-slate-400" />
                      </div>
                    </th>
                    <th
                      className="py-3.5 px-5 cursor-pointer hover:text-slate-900 dark:hover:text-slate-200 transition-colors"
                      onClick={() => handleSort('company')}
                    >
                      <div className="flex items-center gap-1.5">
                        Organization & Website
                        <ArrowUpDown className="w-3.5 h-3.5 text-slate-400" />
                      </div>
                    </th>
                    <th
                      className="py-3.5 px-5 cursor-pointer hover:text-slate-900 dark:hover:text-slate-200 transition-colors"
                      onClick={() => handleSort('status')}
                    >
                      <div className="flex items-center gap-1.5">
                        Status
                        <ArrowUpDown className="w-3.5 h-3.5 text-slate-400" />
                      </div>
                    </th>
                    <th className="py-3.5 px-5 text-right">Actions</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 dark:divide-slate-800 text-sm text-slate-700 dark:text-slate-300">
                  {filteredItems.map((applicant) => (
                    <tr
                      key={applicant.id}
                      className="hover:bg-slate-50/70 dark:hover:bg-slate-800/40 transition-colors group"
                    >
                      <td className="py-4 px-5 whitespace-nowrap">
                        <div className="flex items-center gap-3">
                          <div className="w-9 h-9 rounded-full bg-gradient-to-tr from-[#002045] to-[#1a365d] text-white flex items-center justify-center font-semibold text-sm shadow-xs shrink-0">
                            {applicant.name.charAt(0)}
                          </div>
                          <div>
                            <span className="font-semibold text-slate-900 dark:text-slate-100 block">{applicant.name}</span>
                            <span className="text-xs text-slate-400">@{applicant.username}</span>
                          </div>
                        </div>
                      </td>
                      <td className="py-4 px-5 whitespace-nowrap">
                        <span className="block text-slate-700 dark:text-slate-300">{applicant.email}</span>
                        <span className="text-xs text-slate-400">{applicant.phone}</span>
                      </td>
                      <td className="py-4 px-5 whitespace-nowrap">
                        <div className="flex items-center gap-1.5 text-slate-700 dark:text-slate-300">
                          <Building className="w-3.5 h-3.5 text-slate-400 shrink-0" />
                          <span>{applicant.company?.name || 'N/A'}</span>
                        </div>
                        {applicant.website && (
                          <div className="flex items-center gap-1.5 text-xs text-blue-600 dark:text-blue-400 mt-0.5">
                            <Globe className="w-3.5 h-3.5 shrink-0" />
                            <span>{applicant.website}</span>
                          </div>
                        )}
                      </td>
                      <td className="py-4 px-5 whitespace-nowrap">{getStatusBadge(applicant.status)}</td>
                      <td className="py-4 px-5 whitespace-nowrap text-right">
                        <button className="text-slate-400 hover:text-[#002045] dark:hover:text-blue-400 hover:bg-slate-100 dark:hover:bg-slate-800 p-2 rounded-lg transition-all">
                          <Eye className="w-4 h-4" />
                        </button>
                        <button className="text-slate-400 hover:text-slate-700 dark:hover:text-slate-200 hover:bg-slate-100 dark:hover:bg-slate-800 p-2 rounded-lg transition-all ml-1">
                          <MoreVertical className="w-4 h-4" />
                        </button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
 
            <div className="lg:hidden divide-y divide-slate-100 dark:divide-slate-800">
              {filteredItems.map((applicant) => (
                <div key={applicant.id} className="p-4 space-y-3 bg-white dark:bg-slate-900">
                  <div className="flex items-center justify-between">
                    <div className="flex items-center gap-3">
                      <div className="w-10 h-10 rounded-full bg-gradient-to-tr from-[#002045] to-[#1a365d] text-white flex items-center justify-center font-semibold text-sm shadow-xs shrink-0">
                        {applicant.name.charAt(0)}
                      </div>
                      <div>
                        <h4 className="font-semibold text-slate-900 dark:text-slate-100 text-sm">{applicant.name}</h4>
                        <p className="text-xs text-slate-400">@{applicant.username}</p>
                      </div>
                    </div>
                    {getStatusBadge(applicant.status)}
                  </div>
 
                  <div className="text-xs space-y-1 bg-slate-50 dark:bg-slate-800/50 p-3 rounded-xl">
                    <p className="text-slate-600 dark:text-slate-300">
                      <span className="font-semibold">Email:</span> {applicant.email}
                    </p>
                    <p className="text-slate-600 dark:text-slate-300">
                      <span className="font-semibold">Phone:</span> {applicant.phone}
                    </p>
                    {applicant.company?.name && (
                      <p className="text-slate-600 dark:text-slate-300">
                        <span className="font-semibold">Company:</span> {applicant.company.name}
                      </p>
                    )}
                    {applicant.website && (
                      <p className="text-blue-600 dark:text-blue-400">
                        <span className="font-semibold text-slate-600 dark:text-slate-300">Website:</span> {applicant.website}
                      </p>
                    )}
                  </div>
                </div>
              ))}
            </div>
          </>
        )}
      </div>
    </div>
  );
};
